function createRoom() { playerName = document.getElementById("name").value; roomCode = Math.random().toString(36).substring(2,6).toUpperCase(); // Save host db.ref("rooms/" + roomCode + "/host").set(playerId); db.ref("rooms/" + roomCode + "/players/" + playerId).set(playerName); window.location = "game.html?room=" + roomCode + "&name=" + playerName; } if (roomCode) { db.ref("rooms/" + roomCode + "/players").on("value", snapshot => { const players = snapshot.val() || {}; const playerList = document.getElementById("playerList"); playerList.innerHTML = ""; Object.values(players).forEach(name => { const li = document.createElement("li"); li.textContent = name; playerList.appendChild(li); }); }); db.ref("rooms/" + roomCode + "/host").on("value", snap => { if (snap.exists() && snap.val() === playerId) { document.getElementById("setupBox").style.display = "block"; } }); }
